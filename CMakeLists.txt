# The installation prefix of a Homebrew formula kept out of the default search paths
# (keg-only), or an empty string.
function(wt_brew_prefix formula variable)
  set(${variable} "" PARENT_SCOPE)
  if(APPLE)
    find_program(WT_BREW brew)
    if(WT_BREW)
      execute_process(COMMAND "${WT_BREW}" --prefix "${formula}" OUTPUT_VARIABLE prefix
        OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET RESULT_VARIABLE status)
      if(status EQUAL 0 AND IS_DIRECTORY "${prefix}")
        set(${variable} "${prefix}" PARENT_SCOPE)
      endif()
    endif()
  endif()
endfunction()

  endif()
endif()

function(wt_warnings target)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  if(WT_WERROR)
    target_compile_options(${target} PRIVATE -Werror)
  endif()
endfunction()
foreach(target wt_core wt_engine whisper-transcribator)
  add_dependencies(${target} wt_version)
  wt_warnings(${target})
  if(WT_SANITIZERS)
    target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endif()
endforeach()
