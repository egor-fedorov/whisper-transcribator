# The installation prefix of a Homebrew formula kept out of the default search paths
# (keg-only), or an empty string.
function(wt_brew_prefix formula variable)
  set(${variable} "" PARENT_SCOPE)
  if(APPLE)
    find_program(WT_BREW brew)
    if(WT_BREW)
      execute_process(COMMAND "${WT_BREW}" --prefix "${formula}" OUTPUT_VARIABLE prefix
        OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET RESULT_VARIABLE status)
      if(status EQUAL 0 AND IS_DIRECTORY "${prefix}")
        set(${variable} "${prefix}" PARENT_SCOPE)
      endif()
    endif()
  endif()
endfunction()

function(wt_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /permissive- /utf-8 /Zc:__cplusplus)
    target_compile_definitions(${target} PRIVATE _CRT_SECURE_NO_WARNINGS NOMINMAX _WIN32_WINNT=0x0A00)
    if(WT_WERROR)
      target_compile_options(${target} PRIVATE /WX)
    endif()
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    if(WT_WERROR)
      target_compile_options(${target} PRIVATE -Werror)
    endif()
  endif()
endfunction()
function(wt_manifest target)
  if(WIN32)
    target_sources(${target} PRIVATE "${PROJECT_SOURCE_DIR}/cmake/windows.manifest")
  endif()
endfunction()
