cmake_minimum_required(VERSION 3.22)
project(whisper_transcribator VERSION 0.3.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(WT_CUDA "Build NVIDIA CUDA inference" OFF)
option(WT_SANITIZERS "Instrument application code with ASan/UBSan" OFF)
option(WT_WERROR "Treat warnings in application and tests as errors" OFF)
set(CMAKE_BUILD_RPATH_USE_ORIGIN ON)
set(CMAKE_INSTALL_RPATH "$ORIGIN/../lib;$ORIGIN")
include(cmake/dependencies.cmake)
find_package(PkgConfig REQUIRED)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavformat>=59.27.100 libavcodec>=59.37.100 libswresample>=4.7.100 libavutil>=57.28.100)
find_package(CURL REQUIRED)
find_package(OpenSSL REQUIRED COMPONENTS Crypto)
configure_file(cmake/version.hpp.in generated/version.hpp @ONLY)
configure_file(cmake/package.env.in generated/package.env @ONLY)
file(READ src/model_catalog.json WT_MODEL_CATALOG)
configure_file(cmake/catalog.hpp.in generated/catalog.hpp @ONLY)
add_library(wt_core STATIC src/io.cpp src/jobs.cpp src/models.cpp src/outputs.cpp src/journal.cpp src/pipeline.cpp src/report.cpp)
target_include_directories(wt_core PUBLIC src "${CMAKE_CURRENT_BINARY_DIR}/generated")
target_link_libraries(wt_core PUBLIC CURL::libcurl OpenSSL::Crypto nlohmann_json::nlohmann_json)
add_library(wt_engine STATIC src/audio.cpp src/engine.cpp)
target_link_libraries(wt_engine PUBLIC wt_core whisper PkgConfig::AV)
add_executable(whisper-transcribator src/main.cpp src/cli.cpp)
target_link_libraries(whisper-transcribator PRIVATE wt_engine CLI11::CLI11)
function(wt_warnings target)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  if(WT_WERROR)
    target_compile_options(${target} PRIVATE -Werror)
  endif()
endfunction()
foreach(target wt_core wt_engine whisper-transcribator)
  wt_warnings(${target})
  if(WT_SANITIZERS)
    target_compile_options(${target} PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${target} PUBLIC -fsanitize=address,undefined)
  endif()
endforeach()
install(TARGETS whisper-transcribator RUNTIME DESTINATION bin)
include(CTest)
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
