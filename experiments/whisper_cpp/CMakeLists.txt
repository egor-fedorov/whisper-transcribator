cmake_minimum_required(VERSION 3.22)
project(whisper_transcribator_native VERSION 0.2.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(NATIVE_CUDA "Build NVIDIA CUDA backend" OFF)
option(NATIVE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
set(GGML_CUDA ${NATIVE_CUDA} CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS ON CACHE BOOL "" FORCE)
set(CMAKE_BUILD_RPATH_USE_ORIGIN ON)
set(CMAKE_INSTALL_RPATH "$ORIGIN/../lib;$ORIGIN")
include(FetchContent)
FetchContent_Declare(whisper URL https://github.com/ggml-org/whisper.cpp/archive/927cfce34f31707e17f2bff35c349632fb9e2c3a.tar.gz)
FetchContent_MakeAvailable(whisper)
find_package(PkgConfig REQUIRED)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavformat libavcodec libswresample libavutil)
add_executable(whisper-transcribator-native main.cpp audio.cpp)
target_include_directories(whisper-transcribator-native PRIVATE ${whisper_SOURCE_DIR}/examples)
target_link_libraries(whisper-transcribator-native PRIVATE whisper PkgConfig::AV)
target_compile_options(whisper-transcribator-native PRIVATE -Wall -Wextra -Wpedantic)
if(NATIVE_SANITIZERS)
  target_compile_options(whisper-transcribator-native PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(whisper-transcribator-native PRIVATE -fsanitize=address,undefined)
endif()
install(TARGETS whisper-transcribator-native RUNTIME DESTINATION bin)
enable_testing()
add_test(NAME version COMMAND whisper-transcribator-native --version)
add_test(NAME reject_unknown COMMAND whisper-transcribator-native --compute-type int8)
set_tests_properties(reject_unknown PROPERTIES WILL_FAIL TRUE)
add_test(NAME reject_missing_model COMMAND whisper-transcribator-native --model /missing/model --output-dir /tmp /missing/audio)
set_tests_properties(reject_missing_model PROPERTIES WILL_FAIL TRUE)
